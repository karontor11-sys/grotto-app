def get_iss_daily_log(self, placement_id, log_date):
    """
    Returns the ISS daily log row for a given placement + date, or None if not found.

    log_date can be a datetime.date, datetime.datetime, or YYYY-MM-DD string.
    """
    if log_date is None:
        return None

    # Normalize date to YYYY-MM-DD
    try:
        date_str = log_date.strftime("%Y-%m-%d")  # date/datetime objects
    except Exception:
        date_str = str(log_date)[:10]  # strings

    # Support both connection patterns used in this project
    conn = self.get_connection() if hasattr(self, "get_connection") else self.conn
    cur = conn.cursor()

    # NOTE: If your project uses a different table name, update it here after searching the codebase.
    cur.execute(
        """
        SELECT *
        FROM iss_daily_logs
        WHERE placement_id = ? AND log_date = ?
        LIMIT 1
        """,
        (placement_id, date_str),
    )
    row = cur.fetchone()
    if row is None:
        return None

    # Convert to dict in a way that works for sqlite Row OR tuple
    try:
        return dict(row)  # sqlite3.Row case
    except Exception:
        cols = [d[0] for d in cur.description]
        return dict(zip(cols, row))
