def render_hover_flag_label(label: str, tooltip: str) -> None:
    """
    Renders label text with a hover tooltip (flag) using HTML title attribute.
    Safe for short tooltip text; keeps UI clean in collapsed rows.
    """
    tooltip = (tooltip or "").replace('"', "&quot;")
    st.markdown(
        f'<span title="{tooltip}" style="cursor: help; text-decoration: underline dotted; '

    )
