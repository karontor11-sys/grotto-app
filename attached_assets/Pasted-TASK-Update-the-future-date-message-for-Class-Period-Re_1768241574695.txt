from datetime import date

def _ordinal(n: int) -> str:
    # 11th, 12th, 13th are special cases
    if 10 <= (n % 100) <= 13:
        return "th"
    return {1: "st", 2: "nd", 3: "rd"}.get(n % 10, "th")

def format_date_with_suffix(d: date) -> str:
    mon = d.strftime("%b") + "."
    day = d.day
    return f"{mon} {day}{_ordinal(day)}, {d.year}"
