from datetime import date, datetime, timedelta

def _to_date(d):
    """Safely normalize datetime/date/iso-string -> date (or None)."""
    if d is None:
        return None
    if isinstance(d, date) and not isinstance(d, datetime):
        return d
    if isinstance(d, datetime):
        return d.date()
    if isinstance(d, str):
        try:
            # Accept 'YYYY-MM-DD' and common datetime strings
            return datetime.fromisoformat(d.replace("Z", "")).date()
        except Exception:
            return None
    return None

def _compute_day_of_n(active_day: date, start_day: date, total_days: int):
    """
    Compute "Day X of N" for a placement.
    If anything is missing/invalid, fall back to Day 1 of max(total_days, 1).
    """
    n = max(int(total_days or 1), 1)

    if not active_day or not start_day:
        return 1, n

    # Day number is inclusive from start_day
    delta = (active_day - start_day).days
    day_num = delta + 1

    # Clamp to valid range
    if day_num < 1:
        day_num = 1
    if day_num > n:
        day_num = n

    return day_num, n

def render_lunch_detention_day_progress(placement, active_date):
    """
    Returns the label string for the Lunch Detention dashboard card.
    """
    # Try multiple common field names safely
    start_day = _to_date(getattr(placement, "start_date", None) or placement.get("start_date") if isinstance(placement, dict) else None)
    total_days = getattr(placement, "total_days", None) if not isinstance(placement, dict) else placement.get("total_days")

    # Some codebases store assigned days differently; attempt fallbacks:
    if not total_days:
        # Common alternatives: day_count, num_days, duration_days
        total_days = (
            (getattr(placement, "day_count", None) if not isinstance(placement, dict) else placement.get("day_count"))
            or (getattr(placement, "num_days", None) if not isinstance(placement, dict) else placement.get("num_days"))
            or (getattr(placement, "duration_days", None) if not isinstance(placement, dict) else placement.get("duration_days"))
            or 1
        )

    day_num, n = _compute_day_of_n(_to_date(active_date), start_day, int(total_days))
    return f"📅 Day {day_num} of {n}"
